def cost_function(X, y, beta):    
  m = len(y)    
  return (1/2*m)*np.sum((modele(X,beta) - y)**2)

f) Minimisation of loss function: Gradient Descent

#Gradient function  
def grad_cost_funct(X, y, beta):   
  m = len(y)   
  return (1/m)*(X.T).dot(modele(X,beta) - y)  

#Gradient descent algorithm  
def gradient_descent(X,y,beta, learning_rate, n_iterations):   
  cost_history = np.zeros(n_iterations)   
  for i in range(0, n_iterations):     
    beta = beta - learning_rate*grad_cost_funct(X,y,beta)     
    cost_history[i] = cost_function(X,y,beta)   
    return beta, cost_history

i) Coefficient od determination (R2)

def coeff_determination(y, y_predi):   
  u = ((y-y_predi)**2).sum()   
  v = ((y-np.mean(y))**2).sum()      
  return (1-u/v)  

R2 = coeff_determination(y, y_predi)

2) Model with sklearn

import numpy as np 
import matplotlib.pyplot as plt 
from sklearn.linear_model import LinearRegression 
from sklearn.metrics import mean_squared_error, r2_score  

# Random data 
np.random.seed(0) 
x, y = make_regression(n_samples = 100, n_features = 1, noise = 10)  

# Object initiation
regression = LinearRegression()  

#Training
regression.fit(x,y) 
regression.coef_ 
regression.intercept_  

#Prediction 
y_pred = regression.predict(x)   

#Model evaluation 
R2 = regression.score(x,y) 
rmse = mean_squared_error(y, y_pred) 

# Values printing
print(f"Slope: {regression.coef_}\n ") 
print(f"Intercept: {regression.intercept_}\n RMSE: {rmse} \n R2: {R2}")  

# Plot of original data
plt.close()  
plt.scatter(x,y, s=10) 
plt.xlabel('x') 
plt.ylabel('y') 
plt.show()   

#Predicted value
